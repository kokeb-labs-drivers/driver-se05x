231 static constexpr Reason readLength(ByteSpan in, std::size_t& length,
232 std::size_t& header) noexcept {
233 const uint8_t first = in[0];
234 if (first < 0x80U) {
235 length = first;
236 header = 1U;
237 return Reason::Ok;
238 }
239 if (first == 0x80U) {
240 return Reason::IndefiniteLength;
241 }
242 const auto count = static_cast<std::size_t>(first & 0x7FU);
243 if (count > 2U) {
244 return Reason::LengthTooLarge; // longer than kMaxLength, or the reserved form 0xFF
245 }
246 if (in.size() < (1U + count)) {
247 return Reason::Truncated;
248 }
249 std::size_t value = 0U;
250 for (std::size_t i = 0U; i < count; ++i) {
251 value = (value << 8U) | in[1U + i];
252 }
253 if (LengthFieldSize(value) != (1U + count)) {
254 return Reason::NonMinimalLength; // X.690 10.1: the definite form in the fewest octets
255 }
256 length = value;
257 header = 1U + count;
258 return Reason::Ok;
259 }

370constexpr Reason ParseNamedBits(ByteSpan value, uint16_t& out) noexcept {
371 ByteSpan bits{};
372 uint8_t unused = 0U;
373 const Reason r = ParseBitString(value, bits, unused);
374 if ((r != Reason::Ok) || bits.empty() || (bits.size() > 2U)) {
375 return Reason::BadBitString;
376 }
377 const uint32_t last = bits[bits.size() - 1U];
378 if ((last & (uint32_t{1} << unused)) == 0U) {
379 return Reason::BadBitString; // the last used bit is a trailing zero
380 }
381 uint32_t v = 0U;
382 for (std::size_t i = 0U; i < (bits.size() * 8U); ++i) {
383 const uint32_t byte = bits[i / 8U];

385 v |= uint32_t{1} << i;
386 }
387 }
388 out = static_cast<uint16_t>(v & 0xFFFFU);

470constexpr bool Utf8Lead(uint32_t b, std::size_t& more, uint32_t& cp, uint32_t& min) noexcept {
471 bool ok = true;
472 if ((b > 0U) && (b < 0x80U)) {
473 more = 0U;

475 min = 0U;

477 more = 1U;
478 cp = b & 0x1FU;
479 min = 0x80U;

481 more = 2U;
482 cp = b & 0x0FU;
483 min = 0x800U;
484 } else if ((b & 0xF8U) == 0xF0U) {
485 more = 3U;
486 cp = b & 0x07U;
487 min = 0x10000U;
488 } else {
489 ok = false;

500constexpr Reason CheckUtf8(ByteSpan value) noexcept {
501 std::size_t i = 0U;
502 while (i < value.size()) {
503 std::size_t more = 0U;
504 uint32_t cp = 0U;
505 uint32_t min = 0U;
506 if (!detail::Utf8Lead(value[i], more, cp, min) ||
507 ((more != 0U) && ((i + more) >= value.size()))) {
508 return Reason::BadString;
509 }
510 for (std::size_t k = 1U; k <= more; ++k) {
511 const uint32_t c = value[i + k];
512 if ((c & 0xC0U) != 0x80U) {
513 return Reason::BadString;
514 }
515 cp = (cp << 6U) | (c & 0x3FU);
516 }
517 if ((cp < min) || (cp > 0x10FFFFU) || ((cp >= 0xD800U) && (cp <= 0xDFFFU))) {
518 return Reason::BadString;
519 }
520 i += 1U + more;
521 }
522 return Reason::Ok;
523}

532constexpr Reason ParseTime(const Element& element, Time& out) noexcept {
533 const bool utc = (element.tag == kUtcTime);
534 if (!utc && (element.tag != kGeneralizedTime)) {
535 return Reason::UnexpectedTag;
536 }
537 const ByteSpan v = element.value;
538 const std::size_t digits = utc ? 12U : 14U;
539 if ((v.size() != (digits + 1U)) || (v[digits] != 0x5AU)) {
540 return Reason::BadTime; // the length, or no 'Z'
541 }
542 std::array<uint32_t, 7> f{};
543 for (std::size_t i = 0U; i < digits; ++i) {
544 if ((v[i] < 0x30U) || (v[i] > 0x39U)) {
545 return Reason::BadTime;
546 }
547 f[i / 2U] = (f[i / 2U] * 10U) + (static_cast<uint32_t>(v[i]) - 0x30U);
548 }
549 uint32_t year = 0U;
550 std::size_t at = 1U;
551 if (utc) {
552 year = (f[0] < 50U) ? (2000U + f[0]) : (1900U + f[0]);
553 } else {
554 year = (f[0] * 100U) + f[1];
555 at = 2U;
556 }
557 Time t{};
558 t.year = static_cast<uint16_t>(year & 0xFFFFU);
559 t.month = static_cast<uint8_t>(f[at] & 0xFFU);
560 t.day = static_cast<uint8_t>(f[at + 1U] & 0xFFU);
561 t.hour = static_cast<uint8_t>(f[at + 2U] & 0xFFU);
562 t.minute = static_cast<uint8_t>(f[at + 3U] & 0xFFU);
563 t.second = static_cast<uint8_t>(f[at + 4U] & 0xFFU);
564 if (!IsValid(t)) {
565 return Reason::BadTime;
566 }
567 out = t;
568 return Reason::Ok;
569}

580constexpr Reason ParseEcdsaSignature(ByteSpan der, std::size_t width,
581 MutableByteSpan raw) noexcept {
582 if ((width == 0U) || (raw.size() < (2U * width))) {
583 return Reason::BadSignatureValue;
584 }
585 Reader outer(der, kMaxDepth - 1U);
586 Reader seq(ByteSpan{}, 0U);
587 if ((outer.ExpectEnter(kSequence, seq) != Reason::Ok) || (outer.Finish() != Reason::Ok)) {
588 return Reason::BadSignatureValue;
589 }
590 for (std::size_t k = 0U; k < 2U; ++k) {
591 Element e{};
592 ByteSpan m{};
593 if ((seq.Expect(kInteger, e) != Reason::Ok) || (ParseUnsigned(e.value, m) != Reason::Ok) ||
594 (m.size() > width) || ((m.size() == 1U) && (m[0] == 0U))) {
595 return Reason::BadSignatureValue;
596 }
597 const MutableByteSpan slot = raw.subspan(k * width, width);
598 std::fill(slot.begin(), slot.end(), uint8_t{0});
599 static_cast<void>(std::copy(m.begin(), m.end(),
600 slot.begin() + static_cast<std::ptrdiff_t>(width - m.size())));
601 }
602 return (seq.Finish() == Reason::Ok) ? Reason::Ok : Reason::BadSignatureValue;
603}
