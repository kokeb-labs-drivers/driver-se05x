159template <DigestPort H>
160Reason Digest(H& hasher, SignatureAlgorithm algorithm, ByteSpan message,
161 std::array<uint8_t, kMaxDigest>& digest) noexcept {
162 HashAlg alg = HashAlg::Sha256;
163 if (algorithm == SignatureAlgorithm::EcdsaSha256) {
164 alg = HashAlg::Sha256;
165 } else if (algorithm == SignatureAlgorithm::EcdsaSha384) {
166 alg = HashAlg::Sha384;
167 } else {
168 return Reason::UnsupportedAlgorithm;
169 }
170 return (hasher.Hash(alg, message, MutableByteSpan{digest}) == CryptoStatus::Ok)
171 ? Reason::Ok
172 : Reason::CryptoFailure;
173}

189template <SignatureVerifier V, DigestPort H>
190Reason VerifySigned(V& verifier, H& hasher, const PublicKey& key, SignatureAlgorithm algorithm,
191 ByteSpan message, ByteSpan signature) noexcept {
192 if ((key.type == KeyType::Ed25519) || (algorithm == SignatureAlgorithm::Ed25519)) {
193 return Reason::UnsupportedAlgorithm;
194 }
195 if (AlgorithmFor(key.type) != algorithm) {
196 return Reason::AlgorithmMismatch;
197 }
198 if (!verifier.Supports(key.type)) {
199 return Reason::CurveNotSupported;
200 }
201 if (key.bytes.size() != PublicKeyLength(key.type)) {
202 return Reason::BadPublicKey; // an anchor built by hand with the wrong length
203 }
204 const std::size_t width = DigestLength(algorithm);
205 std::array<uint8_t, kMaxRawSignature> raw{};
206 std::array<uint8_t, kMaxDigest> digest{};
207 Reason r = der::ParseEcdsaSignature(signature, width, MutableByteSpan{raw});
208 r = (r == Reason::Ok) ? Digest(hasher, algorithm, message, digest) : r;
209 if (r != Reason::Ok) {
210 return r;
211 }
212 const CryptoStatus s = verifier.VerifySignature(
213 key.type, key.bytes, ByteSpan{digest}.first(width), ByteSpan{raw}.first(2U * width));
214 Reason out = Reason::CryptoFailure;
215 if (s == CryptoStatus::Ok) {
216 out = Reason::Ok;
217 } else if (s == CryptoStatus::Mismatch) {
218 out = Reason::SignatureInvalid;
219 } else if (s == CryptoStatus::InvalidKey) {
220 out = Reason::BadPublicKey;
221 } else {
222 out = Reason::CryptoFailure;
223 }
224 return out;
225}

242constexpr Reason CheckTime(const Certificate& cert, uint64_t now) noexcept {
243 if (now < ToUnixSeconds(cert.not_before)) {
244 return Reason::NotYetValid;

247 return Reason::Expired;

259constexpr Reason CheckIssuing(const Certificate& cert, Issuer& issuer) noexcept {
260 if (!cert.has_basic_constraints || !cert.is_ca) {
261 return Reason::NotCa;
262 }
263 if (cert.has_key_usage && !HasBits(cert.key_usage, key_usage::kKeyCertSign)) {
264 return Reason::KeyCertSignMissing;
265 }
266 if (issuer.max_path == 0U) {
267 return Reason::PathLengthExceeded;
268 }
269 std::size_t left = issuer.max_path - 1U;
270 if ((cert.path_length != kNoPathLength) && (cert.path_length < left)) {
271 left = cert.path_length;
272 }
273 issuer.name = cert.subject;
274 issuer.key = cert.public_key;
275 issuer.max_path = left;
276 return Reason::Ok;
277}

285constexpr Reason CheckLeaf(const Certificate& cert, const VerifyOptions& options) noexcept {
286 if (cert.is_ca && !options.allow_ca_leaf) {
287 return Reason::LeafIsCa;
288 }
289 const uint16_t need = options.required_key_usage;
290 if ((need != 0U) && (!cert.has_key_usage || !HasBits(cert.key_usage, need))) {
291 return Reason::LeafKeyUsageMissing;
292 }
293 return Reason::Ok;
294}

308template <SignatureVerifier V, DigestPort H>
309Reason Walk(std::span<const ByteSpan> chain, const TrustAnchor& anchor,
310 const VerifyOptions& options, V& verifier, H& hasher, VerifiedChain& out) noexcept {
311 const std::size_t allowed =
312 (anchor.path_length == kNoPathLength) ? kMaxChain : anchor.path_length;
313 Issuer issuer{.name = anchor.subject, .key = anchor.key, .max_path = allowed};
314 for (std::size_t k = 0U; k < chain.size(); ++k) {
315 const std::size_t index = chain.size() - 1U - k;
316 const Result<Certificate> parsed = ParseCertificate(chain[index]);
317 if (!parsed.ok()) {
318 return ReasonOf(parsed.detail);
319 }
320 const Certificate& cert = parsed.value;
321 Reason r = der::SameBytes(cert.issuer, issuer.name) ? Reason::Ok : Reason::NameMismatch;
322 r = (r == Reason::Ok) ? VerifySigned(verifier, hasher, issuer.key, cert.signature_algorithm,
323 cert.tbs, cert.signature)
324 : r;
325 r = (r == Reason::Ok) ? CheckTime(cert, options.now_unix) : r;
326 if (r == Reason::Ok) {
327 r = (index != 0U) ? CheckIssuing(cert, issuer) : CheckLeaf(cert, options);
328 }
329 if (r != Reason::Ok) {
330 return r;
331 }
332 if (index == 0U) {
333 out.leaf_key = cert.public_key;
334 out.leaf_subject = cert.subject;
335 }
336 }
337 out.length = chain.size();
338 return Reason::Ok;
339}

356template <SignatureVerifier V, DigestPort H>
357Result<VerifiedChain> VerifyChain(std::span<const ByteSpan> chain,
358 std::span<const TrustAnchor> anchors,
359 const VerifyOptions& options, V& verifier, H& hasher) noexcept {
360 Reason r = Reason::Ok;
361 if (chain.empty()) {
362 r = Reason::EmptyInput;
363 } else if (chain.size() > kMaxChain) {
364 r = Reason::ChainTooLong;
365 } else if (anchors.size() > kMaxAnchors) {
366 r = Reason::TooManyEntries;
367 } else if (options.now_unix == 0U) {
368 r = Reason::NoTime;
369 } else {
370 r = Reason::Ok;
371 }
372 const Result<Certificate> top =
373 (r == Reason::Ok) ? ParseCertificate(chain[chain.size() - 1U]) : Fail<Certificate>(r);
374 if (!top.ok()) {
375 return Fail<VerifiedChain>(ReasonOf(top.detail));
376 }
377 Reason last = Reason::NoTrustAnchor;
378 for (std::size_t i = 0U; i < anchors.size(); ++i) {
379 if (der::SameBytes(anchors[i].subject, top.value.issuer)) {
380 VerifiedChain out{};
381 last = detail::Walk(chain, anchors[i], options, verifier, hasher, out);
382 if (last == Reason::Ok) {
383 out.anchor_index = i;
384 return Result<VerifiedChain>::Success(out);
385 }
386 }
387 }
388 return Fail<VerifiedChain>(last);
389}
