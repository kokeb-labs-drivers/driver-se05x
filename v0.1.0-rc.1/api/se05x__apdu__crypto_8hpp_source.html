71constexpr Result<void> DecodeTagged(ByteSpan data, const TaggedAnswer& a, std::size_t& out_length,
72 std::size_t& tag_length) noexcept {
73 out_length = 0U;
74 tag_length = 0U;
75 ReplyParser p(data);
76 Result<OptionalTlv> output = Result<OptionalTlv>::Success(OptionalTlv{});
77 if (a.result_tag != kTag1) {
78 output = p.Optional(kTag1);
79 }
80 const Result<ByteSpan> tag = p.Need(a.result_tag);
81 if (!output.ok() || !tag.ok() || !p.End().ok()) {
82 return Fail<void>(Error::FrameError);
83 }
84 if (a.validating) {
85 if ((tag.value.size() != 1U) ||
86 ((tag.value[0] != kResultSuccess) && (tag.value[0] != kResultFailure))) {
87 return Fail<void>(Error::FrameError);
88 }
89 if (tag.value[0] == kResultFailure) {
90 Wipe(a.out, a.out.size());
91 return Fail<void>(Error::AuthFailed);
92 }
93 } else {
94 const Result<void> t = CopyOut(tag.value, a.tag_out, tag_length);
95 if (!t.ok()) {
96 return t;
97 }
98 }
99 return CopyOut(output.value.value, a.out, out_length);
100}
