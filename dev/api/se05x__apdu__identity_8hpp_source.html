57constexpr bool TwoByteTag(ByteSpan data, std::size_t& pos, uint16_t tag, ByteSpan& value) noexcept {
58 if ((pos > data.size()) || ((data.size() - pos) < 3U) ||
59 (BigEndian(data.subspan(pos, 2U)) != static_cast<uint32_t>(tag))) {
60 return false;
61 }
62 TlvReader r(data.subspan(pos + 1U)); // the second tag byte stands in for a one-byte tag
63 const Result<Tlv> t = r.Next();
64 if (!t.ok()) {
65 return false;
66 }
67 const uint8_t first = data[pos + 2U];
68 std::size_t length_bytes = 1U;
69 if (first == 0x81U) {
70 length_bytes = 2U;
71 } else if (first == 0x82U) {
72 length_bytes = 3U;
73 } else {
74 length_bytes = 1U;
75 }
76 value = t.value.value;
77 pos += 2U + length_bytes + value.size();
78 return true;
79}
