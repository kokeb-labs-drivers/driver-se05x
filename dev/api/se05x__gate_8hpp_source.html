238constexpr bool FamilyOf(uint16_t oef, AppletGeneration generation, Family& family) noexcept {

283constexpr Groups GroupsOf(const Capabilities& caps) noexcept {
284 const uint32_t cfg = caps.applet_config;
285 const bool v7 = (caps.generation == AppletGeneration::V7);
286 Groups g{};

288 g.aes = ((cfg & apdu::kConfigAes) != 0U) && (!v7 || ExtBit(caps.ext, 4U, 0x01U));

292 g.hkdf = g.hmac && (!v7 || ExtBit(caps.ext, 5U, 0x01U));

294 g.tls = ((cfg & apdu::kConfigTls) != 0U) && (!v7 || ExtBit(caps.ext, 7U, 0x01U));
295 return g;

305constexpr bool HasKeyNeed(Need need, const Capabilities& caps, const Groups& g) noexcept {
306 const Features& f = caps.features;
307 switch (need) {
308 case Need::EcAny:
309 return g.ecdsa || f.eddsa || f.mont25519;
310 case Need::Ecdsa:
311 return g.ecdsa;
312 case Need::Eddsa:
313 return f.eddsa;
314 case Need::Ecdaa:
315 return f.ecdaa;
316 case Need::Ecdh:
317 return (g.ecdsa || f.mont25519) && !g.ecdh_forbidden;
318 case Need::EcPm:
319 return f.ec_point_multiply && g.ecdsa;
320 case Need::Rsa:
321 return f.rsa;
322 case Need::RsaCrypt:
323 return f.rsa && !g.rsa_crypt_forbidden;
324 case Need::Symm:
325 return g.aes || f.des || g.hmac;
326 case Need::EckaPub:
327 return f.eckey_session_ecka;
328 case Need::Always:
329 case Need::Cipher:
330 case Need::Aead:
331 case Need::Mac:
332 case Need::Digest:
333 case Need::Hkdf:
334 case Need::Pbkdf:
335 case Need::Tls:
336 case Need::Mifare:
337 case Need::I2cm:
338 case Need::SelfTest:
339 default:
340 break;
341 }
342 return false;
343}

574constexpr Result<Capabilities> Derive(const AppletInfo& info, ByteSpan ext, uint16_t oef_id,
575 const ProbeResult& probe) noexcept {
576 const Detail where{
577 .sw = 0U, .command = 0xFFU, .layer = Layer::Device, .step = 0U, .attempts_left = 0xFFU};
578 const uint8_t major = info.version.major;
579 if ((major != 3U) && (major != 6U) && (major != 7U)) {
580 return Result<Capabilities>::Failure(Error::DeviceIdMismatch, where);
581 }
582 Capabilities caps{};
583 caps.applet = info.version;
584 caps.generation = info.version.Generation();
585 const bool ext_ok = (ext.size() == kExtFeatureLength)
586 ? (caps.generation != AppletGeneration::V3)
587 : (ext.empty() && (caps.generation != AppletGeneration::V7));
588 if (!ext_ok) {
589 return Result<Capabilities>::Failure(Error::InvalidArgument, where);
590 }
591 if (!detail::FamilyOf(oef_id, caps.generation, caps.family)) {
592 return Result<Capabilities>::Failure(Error::DeviceIdMismatch, where);
593 }
594 caps.applet_config = info.applet_config;
595 caps.secure_box = info.secure_box;
596 caps.oef_id = oef_id;
597 for (std::size_t i = 0U; i < ext.size(); ++i) {
598 caps.ext[i] = ext[i];
599 }
600 caps.features = detail::FeaturesOf(caps, probe);
601 caps.allowed = detail::AllowedSet(caps);
602 return Result<Capabilities>::Success(caps);
603}

613constexpr Error Check(const Capabilities& caps, const Config& config, Command command) noexcept {
614 if (!se05x::detail::IsEnumerator(command)) {
615 return Error::InvalidArgument;
616 }
617 if (config.held_commands.Contains(command) || !caps.allowed.Contains(command)) {
618 return Error::Unsupported;
619 }
620 if (caps.applet_locked && !detail::InList(detail::kWhileLocked, command)) {
621 return Error::WrongState;
622 }
623 if (caps.platform_scp_required && !caps.platform_scp_active &&
624 !detail::InList(detail::kWithoutScp, command)) {
625 return Error::WrongState;
626 }
627 return Error::Ok;
628}

639constexpr Error CurveAllowed(const Capabilities& caps, EcCurve curve) noexcept {
640 if (!se05x::detail::IsEnumerator(curve)) {
641 return Error::InvalidArgument;
642 }
643 bool ok = (caps.applet_config & apdu::kConfigEcdsa) != 0U;
644 if (curve == EcCurve::Ed25519) {
645 ok = caps.features.eddsa;
646 } else if (curve == EcCurve::Montgomery25519) {
647 ok = caps.features.mont25519;
648 } else if (curve == EcCurve::Montgomery448) {
649 ok = caps.features.mont448;
650 } else if (curve == EcCurve::BnP256) {
651 ok = ok && (caps.generation == AppletGeneration::V3);
652 } else {
653 ok = ok && ((caps.family != Family::Se050F) || (detail::CurveBits(curve) >= 224U));
654 }
655 return ok ? Error::Ok : Error::Unsupported;
656}

665constexpr Error SymmKeyAllowed(const Capabilities& caps, SymmKeyType type) noexcept {
666 const detail::Groups g = detail::GroupsOf(caps);
667 switch (type) {
668 case SymmKeyType::Aes:
669 return g.aes ? Error::Ok : Error::Unsupported;
670 case SymmKeyType::Des:
671 return caps.features.des ? Error::Ok : Error::Unsupported;
672 case SymmKeyType::Hmac:
673 return g.hmac ? Error::Ok : Error::Unsupported;
674 default:
675 break;
676 }
677 return Error::InvalidArgument;
678}
