73 DeviceSigner(Device<P>& device, ObjectId key, KeyType type) noexcept

85 Result<std::size_t> Sign(ByteSpan input, MutableByteSpan signature) noexcept {
86 if (type_ == KeyType::Ed25519) {
87 const auto r = device_->template Execute<Command::EddsaSign>(SignArgs{
88 .key = key_, .algo = EcSignAlgo::Ed25519Pure, .input = input, .signature = signature});
89 return r.ok() ? Result<std::size_t>::Success(r.value.length)
90 : Result<std::size_t>::Failure(r.error, r.detail);
91 }
92 const SignatureAlgorithm algorithm = AlgorithmFor(type_);
93 if (input.size() != DigestLength(algorithm)) {
94 return Fail<std::size_t>(Reason::InvalidInput);
95 }
96 const EcSignAlgo algo =
97 (algorithm == SignatureAlgorithm::EcdsaSha384) ? EcSignAlgo::Sha384 : EcSignAlgo::Sha256;
98 return device_->EcdsaSign(key_, algo, input, signature);
99 }

103 ObjectId key_;
104 KeyType type_;

118template <PortSet P>
119Result<PublicKey> ReadPublicKey(Device<P>& device, ObjectId key, KeyType type,
120 MutableByteSpan out) noexcept {
121 const std::size_t length = PublicKeyLength(type);
122 if ((length == 0U) || (out.size() < length)) {
123 return Fail<PublicKey>((length == 0U) ? Reason::InvalidInput : Reason::BufferTooSmall);
124 }
125 const Result<std::size_t> read = device.ReadObject(key, 0U, out);
126 if (!read.ok()) {
127 return Result<PublicKey>::Failure(read.error, read.detail);
128 }
129 const MutableByteSpan bytes = out.first(std::min(read.value, out.size()));
130 if ((bytes.size() != length) || ((type != KeyType::Ed25519) && (bytes[0] != 0x04U))) {
131 return Fail<PublicKey>(Reason::BadPublicKey);
132 }
133 if (type == KeyType::Ed25519) {
134 wire::ReverseBytes(bytes);
135 }
136 return Result<PublicKey>::Success(PublicKey{.type = type, .bytes = ByteSpan{bytes}});
137}

147template <AppletGeneration G>
148constexpr Result<Policy> CertificatePolicy(AuthObject admin) noexcept {
149 if (admin.id.value == 0U) {
150 return Result<Policy>::Failure(Error::InvalidArgument);
151 }
152 PolicyBuilder<G, ObjectClass::BinaryFile, Profile::Strict> builder{};
153 static_cast<void>(builder.ForAllOthers().template Allow<Rule::Read>());
154 static_cast<void>(builder.For(admin).template Allow<Rule::Read>().template Allow<Rule::Delete>());
155 return builder.Build();
156}

164constexpr Reason CheckReadOnlyPolicy(ByteSpan policy) noexcept {
165 if (policy.empty() || (ValidatePolicyBytes(policy) != Error::Ok)) {
166 return Reason::InvalidInput;
167 }
168 const uint32_t write = policy::HeaderBit(Rule::Write);
169 const uint32_t remove = policy::HeaderBit(Rule::Delete);
170 std::size_t pos = 0U;
171 for (std::size_t n = 0U; (pos < policy.size()) && (n < policy.size()); ++n) {
172 uint32_t auth = 0U;
173 uint32_t header = 0U;
174 for (std::size_t i = 0U; i < 4U; ++i) {
175 auth = (auth << 8U) | policy[pos + 1U + i];
176 header = (header << 8U) | policy[pos + 5U + i];
177 }
178 if (((header & write) != 0U) || ((auth == 0U) && ((header & remove) != 0U))) {
179 return Reason::PolicyAllowsWrite;
180 }
181 pos += 1U + static_cast<std::size_t>(policy[pos]);
182 }
183 return Reason::Ok;
184}

192constexpr std::size_t MaxStoredCertificate(const Policy& policy) noexcept {
193 const std::size_t p = policy.Bytes().size();
194 const std::size_t policy_tlv = 1U + der::LengthFieldSize(p) + p;
195 constexpr std::size_t kFixed = 6U + 4U + 1U + 3U; // TAG_1 id, TAG_3 length, TAG_4 header
196 return kMaxApduPayload - policy_tlv - kFixed;
197}

207constexpr Result<void> CheckReadBack(const Result<std::size_t>& back, ByteSpan scratch,
208 ByteSpan written) noexcept {
209 if (!back.ok()) {
210 return Result<void>::Failure(back.error, back.detail);
211 }
212 const bool same = (back.value == written.size()) && (scratch.size() >= written.size()) &&
213 der::SameBytes(scratch.first(written.size()), written);
214 return same ? Result<void>::Success() : Fail<void>(Reason::ReadBackMismatch);

229template <PortSet P>
230Result<void> StoreCertificate(Device<P>& device, ObjectId id, ByteSpan der, const Policy& policy,
231 MutableByteSpan scratch) noexcept {
232 const Result<Certificate> parsed = ParseCertificate(der);
233 Reason r = parsed.ok() ? CheckReadOnlyPolicy(policy.Bytes()) : ReasonOf(parsed.detail);
234 if ((r == Reason::Ok) && (der.size() > MaxStoredCertificate(policy))) {
235 r = Reason::ObjectTooLarge;
236 }
237 if ((r == Reason::Ok) && (scratch.size() < der.size())) {
238 r = Reason::BufferTooSmall;
239 }
240 if (r != Reason::Ok) {
241 return Fail<void>(r);
242 }
243 const Result<bool> exists = device.ObjectExists(id);
244 if (!exists.ok() || exists.value) {
245 return exists.ok() ? Result<void>::Failure(Error::ObjectExists, Detail{})
246 : Result<void>::Failure(exists.error, exists.detail);
247 }
248 const auto length = static_cast<uint16_t>(der.size() & 0xFFFFU);
249 const Result<void> written = device.WriteBinary(id, policy, 0U, der, length);
250 if (!written.ok()) {
251 return written;
252 }
253 return detail::CheckReadBack(device.ReadObject(id, 0U, scratch.first(der.size())), scratch, der);
254}

265template <PortSet P>
266Result<Certificate> ReadCertificate(Device<P>& device, ObjectId id,
267 MutableByteSpan buffer) noexcept {
268 const Result<std::size_t> read = device.ReadObject(id, 0U, buffer);
269 if (!read.ok()) {
270 return Result<Certificate>::Failure(read.error, read.detail);
271 }
272 const ByteSpan bytes = ByteSpan{buffer}.first(std::min(read.value, buffer.size()));
273 der::Reader reader(bytes);
274 der::Element element{};
275 if ((reader.Expect(der::kSequence, element) != Reason::Ok) || (reader.Finish() != Reason::Ok)) {
276 return Fail<Certificate>(Reason::NotACertificate);
277 }
278 return ParseCertificate(bytes);
279}
