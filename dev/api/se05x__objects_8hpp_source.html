77constexpr bool PolicySetWellFormed(ByteSpan set) noexcept {
78 constexpr std::size_t kMaxPolicies = kMaxPolicyBytes / (kPolicyBase + 1U);
79 std::array<uint32_t, kMaxPolicies + 1U> seen{};
80 std::size_t pos = 0U;
81 std::size_t n = 0U;
82 while ((pos < set.size()) && (n <= kMaxPolicies)) {
83 const std::size_t length = set[pos];
84 if ((length < kPolicyBase) || (length > kPolicyMax) || (length > (set.size() - pos - 1U))) {
85 return false;
86 }
87 const uint32_t auth = detail::ReadBe(set, pos + 1U, 4U);
88 const uint32_t header = detail::ReadBe(set, pos + 5U, 4U);
89 if (((header & kRfuBits) != 0U) || (length != (kPolicyBase + ExtensionLength(header)))) {
90 return false;
91 }
92 for (std::size_t i = 0U; i < n; ++i) {
93 if (seen[i] == auth) {
94 return false;
95 }
96 }
97 seen[n] = auth;
98 ++n;
99 pos += 1U + length;
100 }
101 return (pos == set.size()) && (set.size() <= kMaxPolicyBytes);
102}

115constexpr bool ParseAttributesAs(ByteSpan bytes, std::size_t trailer,
116 ObjectAttributes& out) noexcept {
117 if (bytes.size() < (kAttributeHead + trailer)) {
118 return false;
119 }
120 const std::size_t policy_end = bytes.size() - trailer;
121 const ByteSpan policy = bytes.subspan(kAttributeHead, policy_end - kAttributeHead);
122 const uint8_t indicator = bytes[5];
123 ObjectAttributes a{};
124 if (!apdu::ClassFromWire(bytes[4], a.object_class) ||
125 !apdu::OriginFromWire(bytes[policy_end], a.origin) ||
126 ((indicator != apdu::kSetIndicatorSet) && (indicator != apdu::kSetIndicatorNotSet))) {
127 return false;
128 }
129 if ((policy.size() > a.policy.bytes.size()) || !policy::PolicySetWellFormed(policy)) {
130 return false;
131 }
132 a.authentication = ObjectId{ReadBe(bytes, 8U, 4U)};
133 if (indicator == apdu::kSetIndicatorSet) {
134 const auto failed = static_cast<uint16_t>(ReadBe(bytes, 6U, 2U));
135 a.max_attempts = static_cast<uint16_t>(ReadBe(bytes, 12U, 2U));
136 if (a.max_attempts == 0U) {
137 a.attempts_left = 0xFFFFU; // unlimited
138 } else {
139 a.attempts_left =
140 (failed >= a.max_attempts) ? 0U : static_cast<uint16_t>(a.max_attempts - failed);
141 }
142 }
143 a.version = (trailer == kTrailerV7) ? ReadBe(bytes, policy_end + 1U, 4U) : 0U;
144 for (std::size_t i = 0U; i < policy.size(); ++i) {
145 a.policy.bytes[i] = policy[i];
146 }
147 a.policy.length = policy.size();
148 out = a;
149 return true;
150}
