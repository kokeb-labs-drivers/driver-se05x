99class Cursor {
100public:
102 constexpr explicit Cursor(ByteSpan bytes) noexcept : bytes_(bytes) {}
103
105 [[nodiscard]] constexpr bool Has(std::size_t n) const noexcept {
106 return n <= (bytes_.size() - pos_);
107 }
109 [[nodiscard]] constexpr uint8_t U8() noexcept {
110 if (!Has(1U)) {
111 ok_ = false;
112 return 0U;
113 }
114 const uint8_t v = bytes_[pos_];
115 ++pos_;
116 return v;
117 }
119 [[nodiscard]] constexpr uint16_t U16() noexcept {
120 const auto hi = static_cast<uint32_t>(U8());
121 const auto lo = static_cast<uint32_t>(U8());
122 return static_cast<uint16_t>(((hi << 8U) | lo) & 0xFFFFU);
123 }
125 constexpr void Skip(std::size_t n) noexcept {
126 if (Has(n)) {
127 pos_ += n;
128 } else {
129 ok_ = false;
130 pos_ = bytes_.size();
131 }
132 }
134 [[nodiscard]] constexpr bool Complete() const noexcept {
135 return ok_ && (pos_ == bytes_.size());
136 }
138 [[nodiscard]] constexpr bool Ok() const noexcept {
139 return ok_;
140 }
141
142private:
143 ByteSpan bytes_;
144 std::size_t pos_{0U};
145 bool ok_{true};
146};

155[[nodiscard]] constexpr UsFromMs MsToUs16(uint32_t ms) noexcept {
156 if (ms > kMaxMsInUs16) {
157 return UsFromMs{};
158 }
159 return UsFromMs{.us = static_cast<uint16_t>(ms * 1000U), .ok = true};
160}

163[[nodiscard]] constexpr bool ReadDllp(Cursor& c, uint16_t max_ifsc, LinkParams& p) noexcept {
164 const std::size_t len = c.U8();
165 if ((len < kDllpBytes) || !c.Has(len)) {
166 return false;
167 }
168 p.bwt_ms = c.U16();
169 p.ifsc = c.U16();
170 c.Skip(len - kDllpBytes);
171 return (p.bwt_ms != 0U) && (p.bwt_ms <= kMaxBwtMs) && (p.ifsc != 0U) && (p.ifsc <= max_ifsc);
172}
173
175[[nodiscard]] constexpr bool ReadAtrPlp(Cursor& c, LinkParams& p) noexcept {
176 const std::size_t len = c.U8();
177 if ((len < kAtrPlpBytes) || !c.Has(len)) {
178 return false;
179 }
180 p.max_scl_khz = c.U16();
181 const auto config = static_cast<uint32_t>(c.U8());
182 const UsFromMs mpot = MsToUs16(c.U8());
183 c.Skip(3U); // RFU (1) and RFU (2).
184 p.guard_us = c.U16();
185 p.wakeup_us = c.U16();
186 c.Skip(len - kAtrPlpBytes);
187 p.high_speed = (config & kAtrHighSpeedBit) != 0U;
188 p.stretching = false; // not in the ATR (UM11225 Table 13): set on the chip (AN13015 p. 49)
189 p.min_poll_us = mpot.us;
190 return mpot.ok;
191}
192
194[[nodiscard]] constexpr bool ReadCipPlp(Cursor& c, LinkParams& p) noexcept {
195 const std::size_t len = c.U8();
196 if ((len < kCipPlpBytes) || !c.Has(len)) {
197 return false;
198 }
199 c.Skip(1U); // Configuration: RFU.
200 const UsFromMs pwt = MsToUs16(c.U8());
201 p.max_scl_khz = c.U16();
202 c.Skip(1U); // PST: power saving is requested explicitly by the link.
203 p.min_poll_us = static_cast<uint16_t>(static_cast<uint32_t>(c.U8()) * 100U);
204 p.guard_us = c.U16();
205 c.Skip(len - kCipPlpBytes);
206 p.wakeup_us = pwt.us;
207 return pwt.ok;
208}

211[[nodiscard]] constexpr bool SkipField(Cursor& c, std::size_t max) noexcept {
212 const std::size_t len = c.U8();
213 if ((len > max) || !c.Has(len)) {
214 return false;
215 }
216 c.Skip(len);
217 return true;

229constexpr Result<LinkParams> ParseAtr(ByteSpan atr) noexcept {
230 detail::Cursor c(atr);
231 LinkParams p{};
232 p.flavour = TransportFlavour::Um11225;
233 const uint8_t pver = c.U8();
234 c.Skip(detail::kAtrVidBytes);
235 bool ok = (pver != 0U) && detail::ReadDllp(c, detail::kUmMaxIfs, p);
236 ok = ok && (c.U8() == detail::kPlidI2c);
237 ok = ok && detail::ReadAtrPlp(c, p);
238 ok = ok && detail::SkipField(c, atr.size());
239 if (!ok || !c.Complete()) {
240 return detail::Fail<LinkParams>(Error::FrameError);
241 }
242 p.ifsd = p.ifsc;
243 return Result<LinkParams>::Success(p);
244}

254constexpr Result<LinkParams> ParseCip(ByteSpan cip) noexcept {
255 detail::Cursor c(cip);
256 LinkParams p{};
257 p.flavour = TransportFlavour::Gp;
258 const uint8_t pver = c.U8();
259 const std::size_t iin = c.U8();
260 c.Skip(iin);
261 bool ok = (cip.size() <= detail::kMaxCipBytes) && (pver != 0U) && (iin >= 3U) && (iin <= 4U);
262 ok = ok && (c.U8() == detail::kPlidI2c);
263 ok = ok && detail::ReadCipPlp(c, p);
264 ok = ok && detail::ReadDllp(c, detail::kGpMaxIfs, p);
265 ok = ok && detail::SkipField(c, detail::kMaxCipHistory);
266 if (!ok || !c.Complete()) {
267 return detail::Fail<LinkParams>(Error::FrameError);
268 }
269 p.ifsd = detail::kGpIfsd;
270 return Result<LinkParams>::Success(p);
271}
