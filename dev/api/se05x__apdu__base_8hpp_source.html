296constexpr Result<void> DecodeAttestedV2(ByteSpan data, const AttestedSinks& sinks,
297 AttestedRead& reply) noexcept {
298 ReplyParser p(data);
299 const Result<OptionalTlv> value = p.Optional(kTag1);
300 const Result<ByteSpan> uid = p.NeedLength(kTag2, kUniqueIdLength);
301 const Result<OptionalTlv> attributes = p.Optional(kTag3);
302 const Result<ByteSpan> size = p.NeedLength(kTag4, 2U);
303 const Result<ByteSpan> ts = p.NeedLength(kTagTimestamp, kTimestampLength);
304 const Result<ByteSpan> sig = p.Need(kTagSignature);
305 const Result<void> end = p.End();
306 if (!value.ok() || !uid.ok() || !attributes.ok() || !size.ok() || !ts.ok() || !sig.ok() ||
307 !end.ok()) {
308 return Fail<void>(Error::FrameError);
309 }
310 reply.format = AttestationFormat::V2;
311 reply.object_size = BigEndian(size.value);
312 static_cast<void>(CopyExact(uid.value, reply.unique_id));
313 static_cast<void>(CopyExact(ts.value, reply.timestamp));
314 const Result<void> a = CopyFixed(attributes.value.value, reply.attributes);
315 if (!a.ok()) {
316 return a;
317 }
318 const Result<void> d = CopyOut(value.value.value, sinks.data, reply.data_length);
319 if (!d.ok()) {
320 return d;
321 }
322 return CopyOut(sig.value, sinks.signature, reply.signature_length);
323}

372constexpr Result<void> DecodeAttested(ByteSpan data, const AttestedSinks& sinks,
373 AttestedRead& reply) noexcept {
374 reply = AttestedRead{};
375 TlvReader scan(data);
376 bool v2 = false;
377 for (std::size_t i = 0U; (i < 8U) && !scan.AtEnd(); ++i) {
378 const Result<Tlv> t = scan.Next();
379 if (!t.ok()) {
380 return Fail<void>(Error::FrameError);
381 }
382 v2 = v2 || (t.value.tag == kTagTimestamp) || (t.value.tag == kTagSignature);
383 }
384 return v2 ? DecodeAttestedV2(data, sinks, reply) : DecodeAttestedV1(data, sinks, reply);
385}

395constexpr bool ValidI2cmItem(ByteSpan commands, std::size_t pos, std::size_t item) noexcept {
396 const uint8_t tag = commands[pos];
397 const std::size_t length = BigEndian(commands.subspan(pos + 1U, 2U));
398 const bool fixed_two = (tag == 0x01U) || (tag == 0x04U);
399 if (((tag != 0x01U) && (tag != 0x03U) && (tag != 0x04U)) || (fixed_two && (length != 2U)) ||
400 (length > (commands.size() - pos - 3U)) || ((item == 0U) != (tag == 0x01U))) {
401 return false;
402 }
403 // The clock byte of CONFIGURE: 0x00 100 kHz, 0x01 400 kHz.
404 return (tag != 0x01U) || (commands[pos + 4U] <= 0x01U);
405}

416constexpr bool ValidI2cmCommands(ByteSpan commands) noexcept {
417 if (commands.empty() || (commands.size() > kI2cmCommandMax)) {
418 return false;
419 }
420 std::size_t pos = 0U;
421 for (std::size_t item = 0U; (item < kI2cmCommandMax) && (pos < commands.size()); ++item) {
422 if (((commands.size() - pos) < 3U) || !ValidI2cmItem(commands, pos, item)) {
423 return false;
424 }
425 const uint8_t tag = commands[pos];
426 pos += 3U + BigEndian(commands.subspan(pos + 1U, 2U));
427 if ((tag == 0x04U) && (pos != commands.size())) {
428 return false;
429 }
430 }
431 return pos == commands.size();
432}
