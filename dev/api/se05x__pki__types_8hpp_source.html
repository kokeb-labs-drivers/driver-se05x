165constexpr const char* ToString(Reason reason) noexcept {
166 if (!se05x::detail::IsEnumerator(reason)) {
167 return "Unknown";
168 }
169 return detail::kReasonNames[static_cast<std::size_t>(se05x::detail::ToUnderlying(reason))];
170}

177constexpr Error ErrorOf(Reason reason) noexcept {
178 const auto v = static_cast<uint32_t>(se05x::detail::ToUnderlying(reason));
179 Error error = Error::InvalidArgument;
180 if (reason == Reason::Ok) {
181 error = Error::Ok;
182 } else if ((reason == Reason::UnsupportedVersion) || (reason == Reason::UnsupportedAlgorithm) ||
183 (reason == Reason::UnsupportedCurve) || (reason == Reason::UnknownCriticalExtension) ||
184 (reason == Reason::UniqueIdPresent) || (reason == Reason::CurveNotSupported) ||
185 (reason == Reason::HighTagNumber)) {
186 error = Error::Unsupported;
187 } else if ((v >= static_cast<uint32_t>(Reason::NameMismatch)) &&
188 (v <= static_cast<uint32_t>(Reason::LeafIsCa))) {
189 error = Error::AttestationInvalid;
190 } else if ((reason == Reason::BufferTooSmall) || (reason == Reason::ObjectTooLarge)) {
191 error = Error::BufferTooSmall;
192 } else if ((reason == Reason::CryptoFailure) || (reason == Reason::SignerFailed)) {
193 error = Error::CryptoFailure;
194 } else {
195 error = Error::InvalidArgument;
196 }
197 return error;
198}

206constexpr Reason ReasonOf(const Detail& detail) noexcept {
207 const auto last = static_cast<uint32_t>(se05x::detail::ToUnderlying(Reason::NotACertificate));
208 if ((detail.layer != Layer::Pki) || (static_cast<uint32_t>(detail.step) > last)) {
209 return Reason::Ok;
210 }
211 return static_cast<Reason>(detail.step);
212}

359constexpr bool IsLeapYear(uint32_t year) noexcept {
360 return (((year % 4U) == 0U) && ((year % 100U) != 0U)) || ((year % 400U) == 0U);
361}

368constexpr uint32_t DaysInMonth(uint32_t year, uint32_t month) noexcept {
369 constexpr std::array<uint8_t, 12> kDays{31U, 28U, 31U, 30U, 31U, 30U,
370 31U, 31U, 30U, 31U, 30U, 31U};
371 if ((month == 0U) || (month > 12U)) {
372 return 0U;
373 }
374 const uint32_t extra = ((month == 2U) && IsLeapYear(year)) ? 1U : 0U;
375 return static_cast<uint32_t>(kDays[month - 1U]) + extra;
376}

382constexpr bool IsValid(const Time& t) noexcept {
383 return (t.year >= 1950U) && (t.year <= 9999U) && (t.month >= 1U) && (t.month <= 12U) &&
384 (t.day >= 1U) && (t.day <= DaysInMonth(t.year, t.month)) && (t.hour <= 23U) &&
385 (t.minute <= 59U) && (t.second <= 59U);
386}

394constexpr uint64_t ToUnixSeconds(const Time& t) noexcept {
395 if (!IsValid(t) || (t.year < 1970U)) {
396 return 0U;
397 }
398 uint64_t days = 0U;
399 for (uint32_t y = 1970U; y < t.year; ++y) {
400 days += IsLeapYear(y) ? 366U : 365U;
401 }
402 for (uint32_t m = 1U; m < t.month; ++m) {
403 days += DaysInMonth(t.year, m);
404 }
405 days += static_cast<uint64_t>(t.day) - 1U;
406 return (days * 86400U) + (static_cast<uint64_t>(t.hour) * 3600U) +
407 (static_cast<uint64_t>(t.minute) * 60U) + t.second;
408}

416constexpr int32_t Compare(const Time& a, const Time& b) noexcept {
417 const std::array<uint32_t, 6> x{a.year, a.month, a.day, a.hour, a.minute, a.second};
418 const std::array<uint32_t, 6> y{b.year, b.month, b.day, b.hour, b.minute, b.second};
419 for (std::size_t i = 0U; i < x.size(); ++i) {
420 if (x[i] != y[i]) {
421 return (x[i] < y[i]) ? -1 : 1;
422 }
423 }
424 return 0;
425}
