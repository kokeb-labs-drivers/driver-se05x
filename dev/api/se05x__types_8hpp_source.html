313constexpr bool BusFieldsValid(const Config& config) noexcept {
314 return IsEnumerator(config.flavour) && IsEnumerator(config.scp_policy) &&
315 (config.i2c_address <= 0x7FU) && (config.boot_wait_ms <= kMaxWaitMs) &&
316 (config.atr_poll_budget_ms != 0U) && (config.atr_poll_budget_ms <= kMaxWaitMs) &&
317 (config.poll_interval_us >= 1000U) && (config.poll_interval_us <= 1'000'000U);
318}
320constexpr bool TimeoutsValid(const Timeouts& timeouts) noexcept {
321 const auto in_range = [](uint32_t ms) { return (ms != 0U) && (ms <= kMaxCommandTimeoutMs); };
322 return in_range(timeouts.fast_ms) && in_range(timeouts.key_gen_ms) && in_range(timeouts.kdf_ms) &&
323 in_range(timeouts.attest_ms);
324}
326constexpr bool LimitsValid(const Config& config) noexcept {
327 return (config.frame_retries != 0U) && (config.frame_retries <= kMaxFrameRetries) &&
328 (config.wtx_limit != 0U) && (config.wtx_limit <= kMaxWtxCount) &&
329 (config.auth_attempt_limit != 0U);
330}
331} // namespace detail
