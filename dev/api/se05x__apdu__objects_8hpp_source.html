85constexpr RsaComponents ClassifyRsa(const RsaKeyParts& parts, uint16_t bits) noexcept {
86 RsaComponents c{};
87 c.part = {parts.prime_p, parts.prime_q, parts.exponent_dp, parts.exponent_dq,
88 parts.coefficient_iq, parts.public_exponent, parts.private_exponent, parts.modulus};
89 uint32_t mask = 0U;
90 for (std::size_t i = 0U; i < c.part.size(); ++i) {
91 if (!c.part[i].empty()) {
92 c.part[i] = StripLeadingZeros(c.part[i]);
93 mask |= (1U << i);
94 ++c.count;
95 }
96 }
97 constexpr uint32_t kCrt = 0x1FU; // p q dp dq q^-1
98 constexpr uint32_t kE = 0x20U;
99 constexpr uint32_t kD = 0x40U;
100 constexpr uint32_t kN = 0x80U;
101 c.p2 = kP2Default;
102 if (mask == (kCrt | kE | kN)) {
103 c.key_type = kP1KeyPair;
104 } else if (mask == (kE | kD | kN)) {
105 c.key_type = kP1KeyPair;
106 c.p2 = kP2Raw;
107 } else if (mask == kCrt) {
108 c.key_type = kP1Private;
109 } else if (mask == (kD | kN)) {
110 c.key_type = kP1Private;
111 c.p2 = kP2Raw;
112 } else if (mask == (kE | kN)) {
113 c.key_type = kP1Public;
114 } else {
115 return c;
116 }
117 const std::size_t full = static_cast<std::size_t>(bits) / 8U;
118 bool sizes = IsRsaBits(bits);
119 for (std::size_t i = 0U; i < c.part.size(); ++i) {
120 const std::size_t limit = (i < 5U) ? (full / 2U) : full;
121 sizes = sizes && (c.part[i].size() <= limit);
122 }
123 sizes = sizes && (((mask & kN) == 0U) || (c.part[7].size() == full));
124 c.valid = sizes;
125 return c;
126}

137constexpr bool SymmLengthValid(SymmKeyType type, std::size_t length, bool wrapped) noexcept {
138 if (wrapped && ((length < 16U) || ((length % 8U) != 0U))) {
139 return false;
140 }
141 const std::size_t plain = wrapped ? (length - 8U) : length;
142 switch (type) {
143 case SymmKeyType::Aes:
144 return (plain == 16U) || (plain == 24U) || (plain == 32U);
145 case SymmKeyType::Des:
146 return (plain == 8U) || (plain == 16U) || (plain == 24U);
147 case SymmKeyType::Hmac:
148 return (plain >= 1U) && (plain <= kHmacKeyMax);
149 default:
150 break;
151 }
152 return false;
153}

1572constexpr Error CryptoObjectWire(se05x::CryptoObjectType type, uint8_t algorithm, uint8_t& context,
1573 uint8_t& subtype) noexcept {
1574 if (!CryptoContextWire(type, context)) {
1575 return Error::InvalidArgument;
1576 }
1577 bool ok = false;
1578 switch (type) {
1579 case se05x::CryptoObjectType::Digest:
1580 ok = (algorithm <= static_cast<uint8_t>(DigestMode::Sha512)) &&
1581 DigestWire(static_cast<DigestMode>(algorithm), subtype);
1582 break;
1583 case se05x::CryptoObjectType::Mac:
1584 ok = (algorithm <= static_cast<uint8_t>(MacAlgo::DesCmac8)) &&
1585 MacWire(static_cast<MacAlgo>(algorithm), subtype);
1586 break;
1587 case se05x::CryptoObjectType::Aead:
1588 ok = (algorithm <= static_cast<uint8_t>(AeadMode::AesCcm)) &&
1589 AeadWire(static_cast<AeadMode>(algorithm), subtype);
1590 break;
1591 case se05x::CryptoObjectType::Cipher:
1592 if (algorithm > static_cast<uint8_t>(CipherMode::AesCtr)) {
1593 return Error::InvalidArgument;
1594 }
1595 subtype = CipherModeOf(static_cast<CipherMode>(algorithm)).wire;
1596 return (subtype == 0U) ? Error::Unsupported : Error::Ok;
1597 default:
1598 break;
1599 }
1600 return ok ? Error::Ok : Error::InvalidArgument;
1601}
