389 constexpr CommandSet& Add(Command command) noexcept {
390 if (detail::IsEnumerator(command)) {
391 const auto index = static_cast<std::size_t>(detail::ToUnderlying(command));
392 words_[index / kWordBits] |= (uint32_t{1} << (index % kWordBits));
393 }
394 return *this;
395 }

402 [[nodiscard]] constexpr bool Contains(Command command) const noexcept {
403 if (!detail::IsEnumerator(command)) {
404 return false;
405 }
406 const auto index = static_cast<std::size_t>(detail::ToUnderlying(command));
407 return ((words_[index / kWordBits] >> (index % kWordBits)) & 1U) != 0U;
408 }
409
410private:
411 static constexpr std::size_t kWordBits = 32U;
412 static constexpr std::size_t kWords = (kCommandCount + kWordBits - 1U) / kWordBits;
413 std::array<uint32_t, kWords> words_{};
414};
