130constexpr Reason CheckAttribute(Attribute attribute, ByteSpan value) noexcept {
131 std::size_t max = 0U;
132 static_cast<void>(AttributeOid(attribute, max));
133 if (value.empty() || (value.size() > max)) {
134 return Reason::InvalidInput;
135 }
136 Reason r = Reason::Ok;
137 if (attribute == Attribute::Country) {
138 const auto upper = [](uint8_t b) noexcept { return (b >= 0x41U) && (b <= 0x5AU); };
139 r = std::all_of(value.begin(), value.end(), upper) ? Reason::Ok : Reason::InvalidInput;
140 } else if (attribute == Attribute::SerialNumber) {
141 r = (der::CheckPrintable(value) == Reason::Ok) ? Reason::Ok : Reason::InvalidInput;
142 } else {
143 r = (der::CheckUtf8(value) == Reason::Ok) ? Reason::Ok : Reason::InvalidInput;
144 }
145 return r;

539constexpr Reason CheckRequest(const CsrRequest& request, SignatureAlgorithm algorithm) noexcept {
540 const bool alt = (request.alt_names != nullptr) && (request.alt_names->Count() != 0U);
541 Reason r = Reason::Ok;
542 if ((request.subject == nullptr) || !se05x::detail::IsEnumerator(request.key.type) ||
543 ((static_cast<uint32_t>(request.key_usage) & ~static_cast<uint32_t>(key_usage::kAll)) !=
544 0U)) {
545 r = Reason::InvalidInput;
546 } else if (request.subject->Status() != Reason::Ok) {
547 r = request.subject->Status();
548 } else if ((request.alt_names != nullptr) && (request.alt_names->Status() != Reason::Ok)) {
549 r = request.alt_names->Status();
550 } else if ((request.subject->Count() == 0U) && !alt) {
551 r = Reason::EmptyInput;
552 } else if ((request.key.bytes.size() != PublicKeyLength(request.key.type)) ||
553 ((request.key.type != KeyType::Ed25519) && (request.key.bytes[0] != 0x04U))) {
554 r = Reason::BadPublicKey;
555 } else if (AlgorithmFor(request.key.type) != algorithm) {
556 r = Reason::AlgorithmMismatch;
557 } else {
558 r = Reason::Ok;
559 }
560 return r;
561}

573template <CsrSigner S, DigestPort H>
574Result<std::size_t> SignInfo(S& signer, H& hasher, ByteSpan info,
575 std::array<uint8_t, kMaxSignatureDer>& signature) noexcept {
576 const SignatureAlgorithm algorithm = signer.Algorithm();
577 std::array<uint8_t, kMaxDigest> digest{};
578 ByteSpan input = info;
579 if (algorithm != SignatureAlgorithm::Ed25519) {
580 const Reason d = Digest(hasher, algorithm, info, digest);
581 if (d != Reason::Ok) {
582 return Fail<std::size_t>(d);
583 }
584 input = ByteSpan{digest}.first(DigestLength(algorithm));
585 }
586 const Result<std::size_t> s = signer.Sign(input, MutableByteSpan{signature});
587 if (!s.ok()) {
588 return s;
589 }
590 if ((s.value > signature.size()) ||
591 (CheckSignatureValue(algorithm, ByteSpan{signature}.first(s.value)) != Reason::Ok)) {
592 return Fail<std::size_t>(Reason::SignerOutputInvalid);
593 }
594 return s;
595}

611template <CsrSigner S, DigestPort H>
612Result<std::size_t> BuildCsr(const CsrRequest& request, S& signer, H& hasher,
613 MutableByteSpan out) noexcept {
614 const SignatureAlgorithm algorithm = signer.Algorithm();
615 const Reason check = detail::CheckRequest(request, algorithm);
616 Result<std::size_t> result = Result<std::size_t>::Success(0U);
617 if (check != Reason::Ok) {
618 result = Fail<std::size_t>(check);
619 } else {
620 der::Writer w(out);
621 w.Begin(der::kSequence);
622 w.Begin(der::kSequence);
623 w.Small(0U);
624 if (request.subject != nullptr) {
625 request.subject->Encode(w);
626 }
627 detail::WritePublicKeyInfo(w, request.key);
628 detail::WriteAttributes(w, request);
629 w.End();
630 std::array<uint8_t, kMaxSignatureDer> signature{};
631 const Result<std::size_t> signed_info =
632 (w.Status() == Reason::Ok) ? detail::SignInfo(signer, hasher, w.LastClosed(), signature)
633 : Fail<std::size_t>(w.Status());
634 if (signed_info.ok()) {
635 detail::WriteSignatureAlgorithm(w, algorithm);
636 w.BeginBitString();
637 w.Raw(ByteSpan{signature}.first(signed_info.value));
638 w.End();
639 w.End();
640 result = w.Finish();
641 } else {
642 result = signed_info;
643 }
644 }
645 if (!result.ok()) {
646 std::fill(out.begin(), out.end(), uint8_t{0});
647 }
648 return result;
649}

686inline Reason ParseCsrInfo(const der::Element& element, CsrInfo& out) noexcept {
687 der::Reader outer(element.encoded, 1U);
688 der::Reader info{};
689 der::Element e{};
690 uint32_t version = 1U;
691 bool empty = false;
692 Reason r = outer.ExpectEnter(der::kSequence, info);
693 r = (r == Reason::Ok) ? info.Expect(der::kInteger, e) : r;
694 r = (r == Reason::Ok) ? der::ParseSmall(e.value, 0xFFU, version) : r;
695 if ((r == Reason::Ok) && (version != 0U)) {
696 r = Reason::UnsupportedVersion;
697 }
698 r = (r == Reason::Ok) ? info.Expect(der::kSequence, e) : r;
699 r = (r == Reason::Ok) ? CheckName(e.encoded, empty) : r;
700 out.subject = e.encoded;
701 r = (r == Reason::Ok) ? info.Expect(der::kSequence, e) : r;
702 r = (r == Reason::Ok) ? ParsePublicKeyInfo(e, out.key) : r;
703 der::Reader attributes{};
704 r = (r == Reason::Ok) ? info.Expect(der::ContextConstructed(0U), e) : r;
705 r = (r == Reason::Ok) ? info.Enter(e, attributes) : r;
706 out.attributes = e.value;
707 r = (r == Reason::Ok) ? CheckCsrAttributes(attributes) : r;
708 r = (r == Reason::Ok) ? info.Finish() : r;
709 return (r == Reason::Ok) ? outer.Finish() : r;
710}

725template <SignatureVerifier V, DigestPort H>
726Result<CsrInfo> VerifyCsr(ByteSpan csr, V& verifier, H& hasher) noexcept {
727 CsrInfo out{};
728 der::Reader input(csr);
729 der::Reader top{};
730 der::Element info{};
731 der::Element alg{};
732 der::Element sig{};
733 ByteSpan signature{};
734 Reason r = input.ExpectEnter(der::kSequence, top);
735 r = (r == Reason::Ok) ? input.Finish() : r;
736 r = (r == Reason::Ok) ? top.Expect(der::kSequence, info) : r;
737 r = (r == Reason::Ok) ? detail::ParseCsrInfo(info, out) : r;
738 r = (r == Reason::Ok) ? top.Expect(der::kSequence, alg) : r;
739 r = (r == Reason::Ok) ? ParseSignatureAlgorithm(alg, out.signature_algorithm) : r;
740 r = (r == Reason::Ok) ? top.Expect(der::kBitString, sig) : r;
741 r = (r == Reason::Ok) ? der::ParseOctetBits(sig.value, signature) : r;
742 r = (r == Reason::Ok) ? top.Finish() : r;
743 r = (r == Reason::Ok) ? VerifySigned(verifier, hasher, out.key, out.signature_algorithm,
744 info.encoded, signature)
745 : r;
746 if (r != Reason::Ok) {
747 return Fail<CsrInfo>(r);
748 }
749 return Result<CsrInfo>::Success(out);
750}
