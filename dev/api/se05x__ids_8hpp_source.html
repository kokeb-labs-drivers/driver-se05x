93constexpr bool IsReservedValue(uint32_t id) noexcept {
94 return (id == 0U) || ((id >= kReservedSystemBase) && (id <= kReservedSystemLast)) ||
95 ((id >= kReservedCurveBase) && (id <= kReservedCurveLast)) ||

104constexpr Result<ObjectId> MakeUserObject(uint32_t value) noexcept {
