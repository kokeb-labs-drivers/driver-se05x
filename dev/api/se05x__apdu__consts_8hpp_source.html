719constexpr bool ClassFromWire(uint8_t wire, ObjectClass& object_class) noexcept {
720 constexpr std::array<uint8_t, 16> kWire{0x01U, 0x02U, 0x03U, 0x04U, 0x05U, 0x06U, 0x07U, 0x08U,
721 0x09U, 0x0AU, 0x0BU, 0x0CU, 0x0DU, 0x0FU, 0x10U, 0x11U};
722 constexpr std::array<ObjectClass, 16> kClass{
723 ObjectClass::EcKeyPair, ObjectClass::EcPrivKey, ObjectClass::EcPubKey,
724 ObjectClass::RsaKeyPair, ObjectClass::RsaKeyPair, ObjectClass::RsaPrivKey,
725 ObjectClass::RsaPrivKey, ObjectClass::RsaPubKey, ObjectClass::AesKey,
726 ObjectClass::DesKey, ObjectClass::BinaryFile, ObjectClass::UserId,
727 ObjectClass::Counter, ObjectClass::Pcr, ObjectClass::Curve,
728 ObjectClass::HmacKey};
729 for (std::size_t i = 0U; i < kClass.size(); ++i) {
730 if (kWire[i] == wire) {
731 object_class = kClass[i];
732 return true;
733 }
734 }
735 const uint32_t low = static_cast<uint32_t>(wire) & 0x03U;
736 if ((wire >= 0x21U) && (wire <= 0x73U) && (low != 0U)) {
737 constexpr std::array<ObjectClass, 3> kEc{ObjectClass::EcKeyPair, ObjectClass::EcPrivKey,
738 ObjectClass::EcPubKey};
739 object_class = kEc[low - 1U];
740 return true;
741 }
742 return false;
743}
