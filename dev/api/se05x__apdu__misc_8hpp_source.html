48constexpr bool DfArgsFit(const DfArgs& args, const DfShape& s) noexcept {
49 const bool key_ok = s.uses_key ? (args.key.value != 0U) : (args.key.value == 0U);
50 const bool target_ok = s.uses_target ? (args.target.value != 0U) : (args.target.value == 0U);
51 const bool input_ok = (args.input.size() >= s.input_min) && (args.input.size() <= s.input_max);
52 const bool out_ok = (s.answer == 0U) ? args.out.empty() : (args.out.size() >= s.answer);
53 return key_ok && target_ok && input_ok && out_ok;
54}

64constexpr Result<std::size_t> EncodeDf(MutableByteSpan out, const DfArgs& args,
65 const DfShape& s) noexcept {
66 CommandWriter w(out, kClaPlain, kInsCrypto, kP1Default, s.p2);
67 if (!DfArgsFit(args, s)) {
68 w.Refuse(Error::InvalidArgument);
69 }
70 uint8_t next = kTag1;
71 if (s.uses_key) {
72 w.TlvU32(next, args.key.value);
73 ++next;
74 }
75 if (s.uses_target) {
76 w.TlvU32(next, args.target.value);
77 ++next;
78 }
79 if (s.input_max != 0U) {
80 w.Tlv(next, args.input);
81 }
82 if (s.answer != 0U) {
83 w.ExpectResponse(TlvBudget(s.answer));
84 }
85 return w.Finish();
86}

96constexpr Result<void> DecodeDf(const DfArgs& args, ByteSpan data, const DfShape& s,
97 std::size_t& length) noexcept {
98 length = 0U;
99 if (s.answer == 0U) {
100 return ExpectEmpty(data);
101 }
102 ReplyParser p(data);
103 const Result<ByteSpan> v = p.NeedLength(kTag1, s.answer);
104 if (!v.ok() || !p.End().ok()) {
105 return Fail<void>(Error::FrameError);
106 }
107 return CopyOut(v.value, args.out, length);
108}
