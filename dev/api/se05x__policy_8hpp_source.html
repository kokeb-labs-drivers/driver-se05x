112constexpr uint32_t AllowedRules(AppletGeneration generation, ObjectClass object_class) noexcept {
113 if (!detail::IsEnumerator(object_class) || (object_class == ObjectClass::Curve)) {
114 return 0U;
115 }
116 const bool v7 = (generation == AppletGeneration::V7);
117 uint32_t rules = v7 ? kV7All : kV3All;
118 if (IsSymmetricClass(object_class)) {
119 rules |= v7 ? kV7Symmetric : kV3Symmetric;
120 } else if (IsAsymmetricClass(object_class)) {
121 rules |= v7 ? kV7Asymmetric : kV3Asymmetric;
122 } else if (object_class == ObjectClass::UserId) {
123 rules |= v7 ? 0U : kV3UserId;
124 } else {
125 rules |= v7 ? 0U : kV3Data;
126 }
127 return rules;
128}

212constexpr Error CheckChipRules(const Entry& e, ObjectClass object_class) noexcept {
213 const uint32_t forbid_all = Bit(Rule::ForbidAll);
214 if (((e.allow & e.forbid) != 0U) ||
215 (((e.allow & forbid_all) != 0U) && ((e.allow & ~forbid_all) != 0U))) {
216 return Error::InvalidArgument;
217 }
218 if (((e.allow & Bit(Rule::Read)) != 0U) &&
219 ((object_class == ObjectClass::EcPrivKey) || (object_class == ObjectClass::RsaPrivKey))) {
220 return Error::InvalidArgument; // a private key is never readable (S-03)
221 }
222 if (((e.allow & Bit(Rule::RequirePcrValue)) != 0U) != e.pcr_set) {
223 return Error::PolicyInvalid; // REQUIRE_PCR_VALUE needs its extension, and only it
224 }
225 if ((e.allow & (Bit(Rule::DesfireChangeKey) | Bit(Rule::DerivedInput))) != 0U) {
226 return Error::PolicyInvalid; // their 4-byte extension has no argument here
227 }
228 return Error::Ok;
229}

240constexpr Error CheckEntry(const Entry& e, ObjectClass object_class, uint32_t available,
241 bool strict) noexcept {
242 const Error chip = CheckChipRules(e, object_class);
243 if (chip != Error::Ok) {
244 return chip;
245 }
246 if (!strict || (e.allow == 0U) || ((e.allow & Bit(Rule::ForbidAll)) != 0U)) {
247 return Error::Ok; // a policy that grants nothing needs no explicit refusals
248 }
249 const uint32_t lifecycle = (Bit(Rule::Write) | Bit(Rule::Delete) | Bit(Rule::Gen)) & available;
250 const bool key = IsSymmetricClass(object_class) || IsAsymmetricClass(object_class);
251 if (key && ((e.forbid & lifecycle) != lifecycle)) {
252 return Error::InvalidArgument; // Strict: write, delete and generate forbidden explicitly
253 }
254 if (HoldsSecret(object_class) && (((e.allow & Bit(Rule::RequireSecureMessaging)) == 0U) ||
255 ((e.allow & Bit(Rule::ImportExport)) != 0U))) {
256 return Error::InvalidArgument; // Strict: secure messaging, no export (S-03)
257 }
258 return Error::Ok;
259}

269constexpr Error EncodeEntries(const std::array<Entry, kMaxEntries>& entries, std::size_t count,
270 std::array<uint8_t, kMaxPolicyBytes>& out,
271 std::size_t& length) noexcept {
272 std::size_t pos = 0U;
273 for (std::size_t i = 0U; (i < count) && (i < kMaxEntries); ++i) {
274 uint32_t header = 0U;
275 for (uint32_t r = 0U; r <= static_cast<uint32_t>(Rule::UsageAsHmacPepper); ++r) {
276 if ((entries[i].allow & (uint32_t{1} << r)) != 0U) {
277 header |= HeaderBit(static_cast<Rule>(static_cast<uint8_t>(r)));
278 }
279 }
280 const std::size_t body = kPolicyBase + ExtensionLength(header);
281 if ((pos + 1U + body) > out.size()) {
282 return Error::PolicyInvalid;
283 }
284 const std::array<uint32_t, 2> words{entries[i].auth, header};
285 out[pos] = static_cast<uint8_t>(body);
286 for (std::size_t w = 0U; w < words.size(); ++w) {
287 for (std::size_t b = 0U; b < 4U; ++b) {
288 out[pos + 1U + (4U * w) + b] = static_cast<uint8_t>((words[w] >> (24U - (8U * b))) & 0xFFU);
289 }
290 }
291 if ((header & kPcrBit) != 0U) {
292 for (std::size_t b = 0U; b < 4U; ++b) {
293 out[pos + 9U + b] = static_cast<uint8_t>((entries[i].pcr_id >> (24U - (8U * b))) & 0xFFU);
294 }
295 for (std::size_t b = 0U; b < entries[i].pcr_value.size(); ++b) {
296 out[pos + 13U + b] = entries[i].pcr_value[b];
297 }
298 }
299 pos += 1U + body;
300 }
301 length = pos;
302 return Error::Ok;
303}

321constexpr bool GrantsNone(ByteSpan bytes, uint32_t mask) noexcept {
322 std::size_t at = 0U;
323 bool any = false;
324 while (at < bytes.size()) {
325 const std::size_t length = bytes[at];
326 if ((length < 8U) || ((at + 1U + length) > bytes.size())) {
327 return false;
328 }
329 uint32_t header = 0U;
330 for (std::size_t i = 0U; i < 4U; ++i) {
331 header = (header << 8U) | static_cast<uint32_t>(bytes[at + 5U + i]);
332 }
333 if ((header & mask) != 0U) {
334 return false;
335 }
336 any = true;
337 at += 1U + length;
338 }
339 return any;
340}

549 if (G == AppletGeneration::V6) {
550 return Error::Unsupported;
551 }
552 if (error_ != Error::Ok) {
553 return error_;
554 }

556 return Error::InvalidArgument;
557 }
558 const uint32_t available = policy::AllowedRules(G, C);
559 for (std::size_t i = 0U; i < count_; ++i) {
560 for (std::size_t j = 0U; j < i; ++j) {
561 if (entries_[i].auth == entries_[j].auth) {
562 return Error::PolicyInvalid;
563 }
564 }
565 const Error e = policy::CheckEntry(entries_[i], C, available, P == Profile::Strict);
566 if (e != Error::Ok) {
567 return e;
568 }
569 }
570 return Error::Ok;
571 }

586constexpr Result<FixedBytes<8>> EncodeSessionPolicy(const SessionPolicy& policy) noexcept {
587 FixedBytes<8> out{};
588 uint32_t header = 0U;
589 std::size_t pos = 3U;
590 const std::array<uint16_t, 2> limits{policy.max_apdus, policy.max_time};
591 const std::array<uint32_t, 2> bits{0x8000U, 0x4000U};
592 for (std::size_t i = 0U; i < limits.size(); ++i) {
593 if (limits[i] != 0U) {
594 header |= bits[i];
595 out.bytes[pos] = static_cast<uint8_t>((static_cast<uint32_t>(limits[i]) >> 8U) & 0xFFU);
596 out.bytes[pos + 1U] = static_cast<uint8_t>(static_cast<uint32_t>(limits[i]) & 0xFFU);
597 pos += 2U;
598 }
599 }
600 if (policy.allow_refresh) {
601 header |= 0x2000U;
602 }
603 out.bytes[0] = static_cast<uint8_t>(pos - 1U);
604 out.bytes[1] = static_cast<uint8_t>((header >> 8U) & 0xFFU);
605 out.bytes[2] = static_cast<uint8_t>(header & 0xFFU);
606 out.length = pos;
607 return Result<FixedBytes<8>>::Success(out);
608}
