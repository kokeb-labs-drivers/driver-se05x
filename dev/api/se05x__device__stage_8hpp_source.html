252constexpr ByteSpan ShortestLcForm(ByteSpan command, MutableByteSpan out) noexcept {
253 const std::size_t n = command.size();
254 if ((n <= 7U) || (command[4] != 0U)) {
255 return command;
256 }
257 const std::size_t lc = (static_cast<std::size_t>(command[5]) << 8U) | command[6];
258 if ((lc == 0U) || (lc > 255U) || ((7U + lc) > n) || ((5U + lc) > out.size())) {
259 return command;
260 }
261 for (std::size_t i = 0U; i < 4U; ++i) {
262 out[i] = command[i];
263 }
264 out[4] = static_cast<uint8_t>(lc);
265 for (std::size_t i = 0U; i < lc; ++i) {
266 out[5U + i] = command[7U + i];
267 }
268 return ByteSpan{out.data(), 5U + lc};
269}

278constexpr ByteSpan SignedTlvs(ByteSpan body) noexcept {
279 std::size_t at = 0U;
280 while ((at + 2U) <= body.size()) {
281 if (body[at] == 0x52U) {
282 return body.first(at);
283 }
284 std::size_t length = body[at + 1U];
285 std::size_t head = 2U;
286 if (length == 0x81U) {
287 length = ((at + 3U) <= body.size()) ? body[at + 2U] : body.size();
288 head = 3U;
289 } else if (length == 0x82U) {
290 length = ((at + 4U) <= body.size())
291 ? ((static_cast<std::size_t>(body[at + 2U]) << 8U) | body[at + 3U])
292 : body.size();
293 head = 4U;
294 } else if (length > 0x80U) {
295 return ByteSpan{};
296 } else {
297 // one-byte length
298 }
299 if ((at + head + length) > body.size()) {
300 return ByteSpan{};
301 }
302 at += head + length;
303 }
304 return ByteSpan{};
305}
